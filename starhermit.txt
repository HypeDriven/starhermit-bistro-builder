# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com by hash; never edit it). Re-run `npm run package`
# after every change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.17.0
engine.minor=16
engine.capabilities=platform,storage,clipboard

# Keyboard controls (players rebind keys and pads in the runtime's Options).
# The room is also played with the pointer or touch: tap a table to serve it,
# a stove to pick up, the floor to walk there.
control.table1=Digit1+Numpad1 | Serve table 1
control.table2=Digit2+Numpad2 | Serve table 2
control.table3=Digit3+Numpad3 | Serve table 3
control.table4=Digit4+Numpad4 | Serve table 4
control.table5=Digit5+Numpad5 | Serve table 5
control.table6=Digit6+Numpad6 | Serve table 6
control.table7=Digit7+Numpad7 | Serve table 7
control.table8=Digit8+Numpad8 | Serve table 8
control.table9=Digit9+Numpad9 | Serve table 9
control.pickup=KeyK | Pick up dishes
control.hint=KeyH | Hint
control.undo=KeyU | Rewind (practice)
control.speed=KeyF | Fast-forward
control.camera=KeyC | Frame camera
control.pause=KeyP+Escape | Pause
control.prev_action=BracketLeft | Previous action
control.next_action=BracketRight | Next action
control.commit=Space | Do the chosen action
engine.group.table1=Tables
engine.group.pickup=Service
engine.group.prev_action=Action picker

# Gamepad defaults: the D-pad picks one of the legal actions, A does it.
engine.gamepad.pickup=West
engine.gamepad.hint=North
engine.gamepad.undo=LeftShoulder
engine.gamepad.speed=RightShoulder
engine.gamepad.camera=RightStickPress
engine.gamepad.pause=Start+East
engine.gamepad.prev_action=DpadLeft
engine.gamepad.next_action=DpadRight
engine.gamepad.commit=South
